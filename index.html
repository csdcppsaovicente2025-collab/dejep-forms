// ============================================================ // SISTEMA DEJEP — GERADOR DE ESCALA // Cole este código no Apps Script da planilha de DISPONIBILIDADE // junto com o código do formulário (DEJEP_WebApp.gs) // ============================================================ // CONFIGURAÇÕES — ajuste conforme necessário const CFG = { TOTAL_DEJEPS: 104, SEG_07H: 2, // Servidores na segunda às 07h SEG_11H: 2, // Servidores na segunda às 11h FDS_07H: 4, // Servidores no fim de semana às 07h SEM_23H: 1, // Servidores noturnos (seg a sex) FDS_23H: 1, // Servidores noturnos (sáb e dom) }; // ============================================================ // MENU — aparece no Google Sheets // ============================================================ function onOpen() { SpreadsheetApp.getUi() .createMenu('⚙️ DEJEP') .addItem('Gerar Escala do Mês', 'gerarEscala') .addToUi(); } // ============================================================ // FUNÇÃO PRINCIPAL — GERAR ESCALA // ============================================================ function gerarEscala() { const ss = SpreadsheetApp.getActiveSpreadsheet(); const abaDisp = ss.getSheetByName('Disponibilidade'); if (!abaDisp) { SpreadsheetApp.getUi().alert('Aba "Disponibilidade" não encontrada!'); return; } // Lê dados de disponibilidade const dados = abaDisp.getDataRange().getValues(); if (dados.length < 2) { SpreadsheetApp.getUi().alert('Não há dados de disponibilidade para gerar a escala.'); return; } // Monta mapa de disponibilidade por data // Estrutura: { 'DD/MM/YYYY': [ {nome, regime, turno}, ... ] } const dispPorData = {}; const contadorServidor = {}; // controle de equilíbrio // Cabeçalho: Nome | Dia Semana | Data | Regime | Turno | Obs | Email | Enviado em for (let i = 1; i < dados.length; i++) { const row = dados[i]; const nome = row[0]?.toString().trim(); const data = row[2]?.toString().trim(); const regime = row[3]?.toString().trim(); const turno = row[4]?.toString().trim(); if (!nome || !data) continue; if (!dispPorData[data]) dispPorData[data] = []; dispPorData[data].push({ nome, regime, turno }); if (!contadorServidor[nome]) contadorServidor[nome] = { dejeps: 0, regime, turno }; } // Descobre mês/ano a partir dos dados const todasDatas = Object.keys(dispPorData).sort((a, b) => { return parseData(a) - parseData(b); }); if (todasDatas.length === 0) { SpreadsheetApp.getUi().alert('Nenhuma data encontrada nos dados.'); return; } const primeiraData = parseData(todasDatas[0]); const ano = primeiraData.getFullYear(); const mes = primeiraData.getMonth(); const totalDias = new Date(ano, mes + 1, 0).getDate(); // Gera todos os dias do mês const diasDoMes = []; for (let d = 1; d <= totalDias; d++) { const dt = new Date(ano, mes, d); const fmt = String(d).padStart(2,'0') + '/' + String(mes+1).padStart(2,'0') + '/' + ano; diasDoMes.push({ dt, fmt, diaSemana: dt.getDay() }); } // Monta a escala const escala = {}; // { 'DD/MM/YYYY': { '07h': [...], '11h': [...], '23h': [...] } } let totalUsado = 0; diasDoMes.forEach(({ dt, fmt, diaSemana }) => { escala[fmt] = { '07h': [], '11h': [], '23h': [] }; const dispNesteDia = dispPorData[fmt] || []; const diurnos = dispNesteDia.filter(s => s.turno === 'Diurno'); const noturnos = dispNesteDia.filter(s => s.turno === 'Noturno'); const diaristas = diurnos.filter(s => s.regime === 'Diarista'); const plantonistas = diurnos.filter(s => s.regime === 'Plantonista'); // SEGUNDA-FEIRA if (diaSemana === 1) { // 07h — 2 servidores const sel07 = selecionarServidores(diurnos, CFG.SEG_07H, contadorServidor, false, totalUsado, CFG.TOTAL_DEJEPS); escala[fmt]['07h'] = sel07; sel07.forEach(n => { contadorServidor[n].dejeps++; totalUsado++; }); // 11h — 2 servidores (prioridade plantonistas) const jaEscalados07 = new Set(sel07); const disponiveis11 = diurnos.filter(s => !jaEscalados07.has(s.nome)); const sel11 = selecionarServidores(disponiveis11, CFG.SEG_11H, contadorServidor, false, totalUsado, CFG.TOTAL_DEJEPS, true); escala[fmt]['11h'] = sel11; sel11.forEach(n => { contadorServidor[n].dejeps++; totalUsado++; }); } // TERÇA A SEXTA if (diaSemana >= 2 && diaSemana <= 5) { // 11h — máximo possível, prioridade plantonistas const sel11 = selecionarServidores(diurnos, diurnos.length, contadorServidor, false, totalUsado, CFG.TOTAL_DEJEPS, true); escala[fmt]['11h'] = sel11; sel11.forEach(n => { contadorServidor[n].dejeps++; totalUsado++; }); } // FINS DE SEMANA if (diaSemana === 0 || diaSemana === 6) { // 07h — 4 servidores, prioridade diaristas const sel07 = selecionarServidores(diurnos, CFG.FDS_07H, contadorServidor, true, totalUsado, CFG.TOTAL_DEJEPS); escala[fmt]['07h'] = sel07; sel07.forEach(n => { contadorServidor[n].dejeps++; totalUsado++; }); } // NOTURNO — seg a dom const selNot = selecionarServidores(noturnos, CFG.SEM_23H, contadorServidor, false, totalUsado, CFG.TOTAL_DEJEPS); escala[fmt]['23h'] = selNot; selNot.forEach(n => { contadorServidor[n].dejeps++; totalUsado++; }); }); // Cria planilha nova com a escala criarPlanilhaEscala(escala, diasDoMes, contadorServidor, totalUsado, ano, mes); SpreadsheetApp.getUi().alert( '✅ Escala gerada com sucesso!\n\nTotal de DEJEPs utilizadas: ' + totalUsado + ' / ' + CFG.TOTAL_DEJEPS + '\n\nVerifique a planilha "Escala ' + nomeMes(mes) + ' ' + ano + '" criada no seu Google Drive.' ); } // ============================================================ // SELECIONAR SERVIDORES respeitando equilíbrio e prioridades // ============================================================ function selecionarServidores(lista, quantidade, contador, prioridadeDiarista, totalUsado, totalMax, prioridadePlantonista) { if (totalUsado >= totalMax) return []; const ordenados = [...lista].sort((a, b) => { // Prioridade diarista if (prioridadeDiarista) { if (a.regime === 'Diarista' && b.regime !== 'Diarista') return -1; if (a.regime !== 'Diarista' && b.regime === 'Diarista') return 1; } // Prioridade plantonista if (prioridadePlantonista) { if (a.regime === 'Plantonista' && b.regime !== 'Plantonista') return -1; if (a.regime !== 'Plantonista' && b.regime === 'Plantonista') return 1; } // Equilíbrio: menos DEJEPs primeiro const dejA = contador[a.nome]?.dejeps || 0; const dejB = contador[b.nome]?.dejeps || 0; return dejA - dejB; }); const restante = totalMax - totalUsado; return ordenados.slice(0, Math.min(quantidade, restante)).map(s => s.nome); } // ============================================================ // CRIAR PLANILHA DE ESCALA // ============================================================ function criarPlanilhaEscala(escala, diasDoMes, contador, totalUsado, ano, mes) { const nomePlanilha = 'Escala ' + nomeMes(mes) + ' ' + ano; // Remove se já existir const arquivos = DriveApp.getFilesByName(nomePlanilha); while (arquivos.hasNext()) arquivos.next().setTrashed(true); // Cria nova planilha const nova = SpreadsheetApp.create(nomePlanilha); const aba = nova.getActiveSheet(); aba.setName('Escala'); const DIAS_SEMANA = ['Dom','Seg','Ter','Qua','Qui','Sex','Sáb']; const COR_TITULO = '#1a3a5c'; const COR_CABECALHO = '#2d6a9f'; const COR_07H = '#d9ead3'; const COR_11H = '#fce5cd'; const COR_23H = '#cfe2f3'; // Título aba.getRange(1, 1, 1, 8).merge() .setValue('DEJEP — ' + nomeMes(mes).toUpperCase() + ' ' + ano + ' — CPP SÃO VICENTE') .setBackground(COR_TITULO).setFontColor('#ffffff') .setFontSize(14).setFontWeight('bold').setHorizontalAlignment('center'); aba.getRange(2, 1, 1, 8).merge() .setValue('Total de DEJEPs utilizadas: ' + totalUsado + ' / 104') .setFontStyle('italic').setFontColor('#555555').setHorizontalAlignment('center'); let linha = 4; // Agrupa por semana let semana = []; diasDoMes.forEach(({ dt, fmt, diaSemana }, idx) => { semana.push({ dt, fmt, diaSemana }); const proxIdx = idx + 1; const fimMes = proxIdx >= diasDoMes.length; const proxDomingo = !fimMes && diasDoMes[proxIdx].diaSemana === 0; if (fimMes || proxDomingo) { // Cabeçalho da semana const cabRow = ['Horário']; semana.forEach(d => cabRow.push(DIAS_SEMANA[d.diaSemana] + ' ' + d.fmt.substring(0,5))); while (cabRow.length < 8) cabRow.push(''); aba.getRange(linha, 1, 1, 8).setValues([cabRow]) .setBackground(COR_CABECALHO).setFontColor('#ffffff') .setFontWeight('bold').setHorizontalAlignment('center'); linha++; // Linhas de horário [ { chave: '07h', label: '07 às 15h', cor: COR_07H }, { chave: '11h', label: '11 às 19h', cor: COR_11H }, { chave: '23h', label: '23 às 07h', cor: COR_23H }, ].forEach(turno => { let maxServ = 1; semana.forEach(d => { const lista = escala[d.fmt]?.[turno.chave] || []; if (lista.length > maxServ) maxServ = lista.length; }); for (let i = 0; i < maxServ; i++) { const row = [i === 0 ? turno.label : '']; semana.forEach(d => { const lista = escala[d.fmt]?.[turno.chave] || []; row.push(lista[i] || ''); }); while (row.length < 8) row.push(''); aba.getRange(linha, 1, 1, 8).setValues([row]).setBackground(turno.cor); if (i === 0) aba.getRange(linha, 1).setFontWeight('bold'); linha++; } }); aba.getRange(linha - (semana.length > 0 ? 4 : 1), 1, 4, 8) .setBorder(true, true, true, true, true, true, '#cccccc', SpreadsheetApp.BorderStyle.SOLID); semana = []; linha++; } }); // Resumo por servidor linha += 1; aba.getRange(linha, 1, 1, 4).merge() .setValue('RESUMO POR SERVIDOR') .setBackground(COR_TITULO).setFontColor('#ffffff').setFontWeight('bold'); linha++; aba.getRange(linha, 1, 1, 4).setValues([['Servidor', 'Regime', 'Turno', 'Qtd DEJEPs']]) .setBackground(COR_CABECALHO).setFontColor('#ffffff').setFontWeight('bold'); linha++; Object.entries(contador) .filter(([_, s]) => s.dejeps > 0) .sort((a, b) => b[1].dejeps - a[1].dejeps) .forEach(([nome, s], i) => { aba.getRange(linha, 1, 1, 4).setValues([[nome, s.regime, s.turno, s.dejeps]]); if (i % 2 === 0) aba.getRange(linha, 1, 1, 4).setBackground('#f0f4f8'); linha++; }); // Formatação de colunas aba.setColumnWidth(1, 110); for (let c = 2; c <= 8; c++) aba.setColumnWidth(c, 200); // Link para a planilha Logger.log('Planilha criada: ' + nova.getUrl()); } // ============================================================ // AUXILIARES // ============================================================ function parseData(dataStr) { const [d, m, a] = dataStr.split('/').map(Number); return new Date(a, m - 1, d); } function nomeMes(mes) { return ['Janeiro','Fevereiro','Março','Abril','Maio','Junho', 'Julho','Agosto','Setembro','Outubro','Novembro','Dezembro'][mes]; }
