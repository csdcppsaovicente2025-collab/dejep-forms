// ============================================================ // APPS SCRIPT — RECEBE DADOS DO FORMULÁRIO WEB E SALVA NA PLANILHA // Cole em: Extensões > Apps Script da sua planilha Google Sheets // ============================================================ const SPREADSHEET_ID = 'COLE_O_ID_DA_SUA_PLANILHA_AQUI'; const ABA_DISPONIBILIDADE = 'Disponibilidade'; function doPost(e) { try { const dados = JSON.parse(e.postData.contents); const ss = SpreadsheetApp.openById(SPREADSHEET_ID); const aba = ss.getSheetByName(ABA_DISPONIBILIDADE); if (!aba) return resposta(false, 'Aba Disponibilidade não encontrada.'); const { nome, email, regime, obs, linhas } = dados; if (!nome || !regime || !linhas || linhas.length === 0) return resposta(false, 'Dados incompletos.'); const agora = Utilities.formatDate(new Date(), Session.getScriptTimeZone(), 'dd/MM/yyyy HH:mm'); // Garante cabeçalho na linha 3 (padrão da planilha DEJEP) garantirCabecalho(aba); // Salva cada dia como uma linha // Colunas: Nome | Dia da semana | Data | Regime | Observação | E-mail | Enviado em linhas.forEach(linha => { aba.appendRow([ nome, linha.dow, linha.data, regime, obs || '', email || '', agora ]); }); // Envia comprovante por e-mail if (email && email.includes('@')) { enviarComprovante(nome, email, regime, obs, linhas, agora); } return resposta(true, 'Disponibilidade registrada com sucesso!'); } catch (err) { return resposta(false, 'Erro interno: ' + err.message); } } function garantirCabecalho(aba) { const val = aba.getRange('A3').getValue(); if (val !== 'Nome Completo') { aba.getRange('A3:G3').setValues([[ 'Nome Completo', 'Dia da Semana', 'Data', 'Regime', 'Observação', 'E-mail', 'Enviado em' ]]); aba.getRange('A3:G3') .setBackground('#1a3a5c') .setFontColor('#ffffff') .setFontWeight('bold'); } } function enviarComprovante(nome, email, regime, obs, linhas, agora) { const primeiroNome = nome.split(' ')[0]; let tabelaLinhas = ''; linhas.forEach(l => { tabelaLinhas += ` ${l.data} ${l.dow} `; }); const obsBloco = obs ? `
Observação informada:
${obs}
` : ''; const html = `
Polícia Penal SP
CPP São Vicente
Comprovante de Disponibilidade DEJEP

Olá, ${primeiroNome}! Sua disponibilidade foi registrada com sucesso.

Nome${nome}
Regime${regime}
Enviado em${agora}

Dias disponíveis informados (${linhas.length} dia${linhas.length > 1 ? 's' : ''})

${tabelaLinhas}
Data Dia da semana
${obsBloco}
Atenção: Este e-mail confirma que sua disponibilidade foi recebida. A escala final será definida pela coordenação e comunicada posteriormente.
CPP São Vicente • Sistema de Escala DEJEP • E-mail automático
`; GmailApp.sendEmail(email, '✅ Comprovante DEJEP — CPP São Vicente', '', { htmlBody: html, name: 'DEJEP CPP São Vicente' }); } function resposta(sucesso, mensagem) { return ContentService .createTextOutput(JSON.stringify({ sucesso, mensagem })) .setMimeType(ContentService.MimeType.JSON); } // ============================================================ // COMO PUBLICAR: // 1.
